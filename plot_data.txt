def read_flux(filename):
    with open(filename) as file:
        lines = [l.strip() for l in file if l.strip()]
    flux = []
    for line in lines:
        line = line.split(" ")[0]
        if '-' in line or '0.00' in line or 'nan' in line:
            flux.append(np.nan)
        else:
            flux.append(float(line))
    return np.asarray(flux)
